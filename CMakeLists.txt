cmake_minimum_required(VERSION 3.20)
project(GotchaForceNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
include(CTest)
option(GF_BUILD_DESKTOP "Build the SDL2/OpenGL desktop host" ON)
option(GF_BUILD_DEVICE_TESTS "Build tests that open graphics/input/audio devices" OFF)
option(GF_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
add_library(gf_core INTERFACE)
target_include_directories(gf_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gf_core INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(GF_SANITIZERS)
    target_compile_options(gf_core INTERFACE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
    target_link_options(gf_core INTERFACE -fsanitize=address,undefined,float-cast-overflow)
  endif()
endif()
add_library(gf_native_storage STATIC platform/pc/account_save_storage.cpp)
find_package(Threads REQUIRED)
target_link_libraries(gf_native_storage PUBLIC gf_core Threads::Threads)
if(GF_BUILD_DESKTOP OR GF_BUILD_DEVICE_TESTS)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  find_package(OpenGL REQUIRED)
  add_library(gf_platform_pc STATIC platform/pc/sdl_platform.cpp platform/gl/gl_renderer.cpp)
  target_link_libraries(gf_platform_pc PUBLIC gf_core gf_native_storage PkgConfig::SDL2 OpenGL::GL)
endif()
if(GF_BUILD_DESKTOP)
  add_executable(gf_native platform/pc/native_main.cpp)
  target_link_libraries(gf_native PRIVATE gf_platform_pc)
  if(BUILD_TESTING)
    add_test(NAME native_headless COMMAND gf_native --headless --frames 3)
    set_tests_properties(native_headless PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endif()
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  file(GLOB NATIVE_TESTS CONFIGURE_DEPENDS tests/unit/*.cpp)
  foreach(source IN LISTS NATIVE_TESTS)
    get_filename_component(name "${source}" NAME_WE)
    file(READ "${source}" body)
    set(device_test OFF)
    if(body MATCHES "platform/pc/sdl_platform.hpp|platform/gl/gl_renderer.hpp|SDL.h|SDL2/")
      set(device_test ON)
    endif()
    if(device_test AND NOT GF_BUILD_DEVICE_TESTS)
      continue()
    endif()
    add_executable(test_${name} "${source}")
    target_link_libraries(test_${name} PRIVATE gf_core gf_native_storage)
    if(device_test)
      target_link_libraries(test_${name} PRIVATE gf_platform_pc)
    endif()
    if(name STREQUAL "sdl_outputs")
      target_link_options(test_${name} PRIVATE -Wl,--wrap=SDL_GameControllerRumble)
    endif()
    if(name STREQUAL "account_save_storage")
      target_link_options(test_${name} PRIVATE -Wl,--wrap=fsync)
    endif()
    if(name MATCHES "^account_save_storage(_edges)?$")
      add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/run_account_save_storage.py $<TARGET_FILE:test_${name}>)
    elseif(name STREQUAL "startup_sound_timebase" OR name STREQUAL "render_copy_compat_gl")
      add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/run_output_test.py $<TARGET_FILE:test_${name}>)
    else()
      add_test(NAME ${name} COMMAND test_${name})
    endif()
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} SKIP_RETURN_CODE 77)
  endforeach()
  add_test(NAME extraction_tests COMMAND ${Python3_EXECUTABLE} -B -m unittest discover -s tests/unit -p "test_*.py")
  set_tests_properties(extraction_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
